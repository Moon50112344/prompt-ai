  function compareIp(a,b){

    var aa=
      a.split('.')
       .map(
         function(x){
           return parseInt(x)||0;
         }
       );

    var bb=
      b.split('.')
       .map(
         function(x){
           return parseInt(x)||0;
         }
       );


    for(
      var i=0;
      i<Math.min(aa.length,bb.length);
      i++
    ){

      if(aa[i]!==bb[i])
        return aa[i]-bb[i];

    }


    return aa.length-bb.length;

  }

      stopScanner();
